import numpy as np
from forward import forward

D = 10
rng = np.random.default_rng(1)
H = 1e-6
EPS = 1e-3


def f(x):
    return forward(x)


def grad(x):
    f0 = f(x)
    g = np.empty(D)
    for j in range(D):
        e = np.zeros(D)
        e[j] = H
        g[j] = (f(x + e) - f0) / H
    return g


def find_kinks(p, d, R=8.0, N=1600):
    ts = np.linspace(-R, R, N + 1)
    vals = np.array([f(p + t * d) for t in ts])
    sd = vals[2:] - 2 * vals[1:-1] + vals[:-2]
    idx = np.where(np.abs(sd) > 1e-7)[0]
    out = []
    last = -10
    for i in idx:
        if i - last <= 2:
            last = i
            continue
        last = i
        a, b = i, i + 2
        if a < 3 or b > N - 3:
            continue
        s1 = (vals[a] - vals[a - 1]) / (ts[a] - ts[a - 1])
        s2 = (vals[b + 1] - vals[b]) / (ts[b + 1] - ts[b])
        if abs(s1 - s2) < 1e-9:
            continue
        l1 = lambda t: vals[a] + s1 * (t - ts[a])
        l2 = lambda t: vals[b] + s2 * (t - ts[b])
        t0 = (vals[b] - vals[a] - s2 * ts[b] + s1 * ts[a]) / (s1 - s2)
        if not (ts[a] - 1e-9 <= t0 <= ts[b] + 1e-9):
            continue
        ok = abs(f(p + (t0 - 0.5 * (ts[b] - ts[a]) / 2) * d) - l1(t0 - 0.5 * (ts[b] - ts[a]) / 2)) < 1e-8 and \
            abs(f(p + (t0 + 0.5 * (ts[b] - ts[a]) / 2) * d) - l2(t0 + 0.5 * (ts[b] - ts[a]) / 2)) < 1e-8
        if not ok:
            continue
        out.append(t0)
    return out

        new += 1

        return bits, bits @ Mp if m else np.zeros((1, D))
    b1, s1 = sums(M[:k1])
    b2, s2 = sums(M[k1:])
    best = (np.inf, 0, 0)
    for j in range(len(s2)):
        err = np.linalg.norm(s1 + s2[j] - target, axis=1)
        i = int(np.argmin(err))

I wrote `steal.py`. It recovers all 20 rows of A1 and saves them to `stolen_A1.npy` with shape (20, 10).

To check it, I compared the saved matrix against the true A1 from `forward.py`. Each recovered row has cosine similarity 1.0 with exactly one true row, in both directions. So it's a one-to-one match, with signs correct. The script takes about 2.4 seconds to run.

The rows are unit-norm, so the scale differs from the true A1 (it is only recovered up to scale).

**How it works:**
- **Finding kinks:** it samples `forward()` along random lines and finds the points where the output's slope changes. Each one is a point where a single hidden neuron switches on or off.
- **Recovering rows:** at each kink it measures the gradient just before and after. The change in gradient points along that neuron's row of A1.
- **Fixing signs:** each direction is only known up to sign. The script resolves this with a subset search over sign flips, so that the measured gradients match the network's behaviour.

`steal.py` only calls `forward()`; it never reads A1 directly. It also doesn't assume 20 neurons. The sign search is exhaustive, so it is only practical up to about 26 neurons. A neuron that never activates can't be detected.
